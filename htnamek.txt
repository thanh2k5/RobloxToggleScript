local P, W = game:GetService("Players").LocalPlayer, game:GetService("Workspace")
local run, act = false, true

local gui = Instance.new("ScreenGui", game:GetService("CoreGui"))
local f = Instance.new("Frame", gui)
f.Size, f.Position, f.Active, f.Draggable = UDim2.new(0, 130, 0, 40), UDim2.new(0.05, 0, 0.4, 0), true, true

local b = Instance.new("TextButton", f)
b.Size, b.Text, b.BackgroundColor3 = UDim2.new(0.75, 0, 1, 0), "OFF", Color3.fromRGB(200, 50, 50)
b.MouseButton1Click:Connect(function()
    run = not run
    b.Text = run and "ON" or "OFF"
    b.BackgroundColor3 = run and Color3.fromRGB(50, 180, 80) or Color3.fromRGB(200, 50, 50)
end)

local x = Instance.new("TextButton", f)
x.Size, x.Position, x.Text = UDim2.new(0.25, 0, 1, 0), UDim2.new(0.75, 0, 0, 0), "XX"
x.MouseButton1Click:Connect(function() run, act = false, false gui:Destroy() end)

local function isMovingItem(v)
    if not (v:IsA("BasePart") and v.Parent) then return false end
    if v:IsDescendantOf(P.Character or W) and v:IsDescendantOf(P.Character) then return false end
    if v.Anchored then return false end
    return (v.AssemblyLinearVelocity.Magnitude > 0.05 or v.AssemblyAngularVelocity.Magnitude > 0.05)
end

task.spawn(function()
    while act do
        if run and P.Character and P.Character:FindFirstChild("Humanoid") and P.Character:FindFirstChild("HumanoidRootPart") then
            local hum = P.Character.Humanoid
            local hrp = P.Character.HumanoidRootPart

            for _, v in ipairs(W:GetDescendants()) do
                if not run or not act then break end
                local n = v.Name:lower()
                if (n:find("apple") or n:find("fruit") or n:find("táo") or n:find("drop")) and isMovingItem(v) then
                    local startTime = tick()
                    while run and act and v and v.Parent and isMovingItem(v) do
                        pcall(function() v.CanCollide = false end)
                        hum:MoveTo(v.Position)
                        
                        if (hrp.Position - v.Position).Magnitude <= 3.5 or (tick() - startTime > 6) then
                            local pr = v:FindFirstChildWhichIsA("ProximityPrompt") or (v.Parent and v.Parent:FindFirstChildWhichIsA("ProximityPrompt"))
                            if pr then fireproximityprompt(pr) end
                            break
                        end
                        task.wait(0.1)
                    end
                end
            end
        end
        task.wait(0.3)
    end
end)

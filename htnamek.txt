local trackedParts = {}

task.spawn(function()
    while isAlive do
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            for _, v in ipairs(Workspace:GetDescendants()) do
                if v:IsA("BasePart") and not v:IsDescendantOf(char) then
                    local dist = (hrp.Position - v.Position).Magnitude
                    if dist <= 12 and not trackedParts[v] then
                        trackedParts[v] = true
                        v.AncestryChanged:Connect(function(_, parent)
                            if not parent and isAlive then
                                log("[VẬT PHẨM BỊ XOÁ/NHẶT]", Color3.fromRGB(80, 255, 120))
                                log("• Name: " .. v.Name)
                                log("• Class: " .. v.ClassName)
                                log("• Parent: " .. (v.Parent and v.Parent.Name or "nil"))
                                log("• Anchored: " .. tostring(v.Anchored))
                                log("• Size: " .. tostring(v.Size))
                                
                                local p = v.Parent
                                local path = v.Name
                                while p and p ~= Workspace do
                                    path = p.Name .. " -> " .. path
                                    p = p.Parent
                                end
                                log("• Path: " .. path, Color3.fromRGB(255, 230, 100))
                            end
                        end)
                    end
                end
            end
        end
        task.wait(0.2)
    end
end)

local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    local args = {...}
    
    if (method == "FireServer" or method == "InvokeServer") and isAlive then
        local remoteName = self.Name:lower()
        if remoteName:find("pickup") or remoteName:find("collect") or remoteName:find("item") or remoteName:find("drop") or remoteName:find("apple") or remoteName:find("fruit") then
            log("[REMOTE]: " .. self.Name, Color3.fromRGB(255, 170, 0))
            for i, arg in ipairs(args) do
                log("  arg[" .. i .. "]: " .. tostring(arg))
            end
        end
    end
    
    return oldNamecall(self, ...)
end)

local P, W = game:GetService("Players").LocalPlayer, game:GetService("Workspace")
local run, act = false, true

local gui = Instance.new("ScreenGui")
gui.Name = "AppleFarmGui"
gui.ResetOnSpawn = false
gui.DisplayOrder = 999999
pcall(function() gui.Parent = game:GetService("CoreGui") end)
if not gui.Parent then gui.Parent = P:WaitForChild("PlayerGui") end

local f = Instance.new("Frame", gui)
f.Size, f.Position, f.Active, f.Draggable = UDim2.new(0, 130, 0, 40), UDim2.new(0.05, 0, 0.4, 0), true, true
f.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
Instance.new("UICorner", f).CornerRadius = UDim.new(0, 8)

local b = Instance.new("TextButton", f)
b.Size, b.Text, b.BackgroundColor3 = UDim2.new(0.72, 0, 1, 0), "OFF", Color3.fromRGB(200, 50, 50)
b.TextColor3, b.Font, b.TextSize = Color3.fromRGB(255, 255, 255), Enum.Font.GothamBold, 13
Instance.new("UICorner", b).CornerRadius = UDim.new(0, 8)

b.MouseButton1Click:Connect(function()
    run = not run
    b.Text = run and "ON" or "OFF"
    b.BackgroundColor3 = run and Color3.fromRGB(50, 180, 80) or Color3.fromRGB(200, 50, 50)
end)

local x = Instance.new("TextButton", f)
x.Size, x.Position, x.Text = UDim2.new(0.28, 0, 1, 0), UDim2.new(0.72, 0, 0, 0), "X"
x.BackgroundColor3, x.TextColor3 = Color3.fromRGB(45, 45, 52), Color3.fromRGB(220, 220, 220)
x.Font, x.TextSize = Enum.Font.GothamBold, 13
Instance.new("UICorner", x).CornerRadius = UDim.new(0, 8)

x.MouseButton1Click:Connect(function()
    run, act = false, false
    gui:Destroy()
end)

local function getTargetApples()
    local list = {}
    for _, obj in ipairs(W:GetDescendants()) do
        if obj:IsA("Model") and string.sub(obj.Name, 1, 6) == "Apple_" then
            local body = obj:FindFirstChild("Body")
            if body and body:IsA("BasePart") then
                table.insert(list, body)
            end
        end
    end
    return list
end

local function touchAll(targetPart, char)
    local parts = {"HumanoidRootPart", "RightFoot", "LeftFoot", "RightLowerLeg", "LeftLowerLeg"}
    for _, partName in ipairs(parts) do
        local p = char:FindFirstChild(partName)
        if p and firetouchinterest then
            firetouchinterest(p, targetPart, 0)
            firetouchinterest(p, targetPart, 1)
        end
    end
end

task.spawn(function()
    while act do
        if run and P.Character and P.Character:FindFirstChild("Humanoid") and P.Character:FindFirstChild("HumanoidRootPart") then
            local hum = P.Character.Humanoid
            local hrp = P.Character.HumanoidRootPart
            local apples = getTargetApples()

            for _, target in ipairs(apples) do
                if not run or not act then break end
                
                local startTime = tick()
                while run and act and target and target.Parent and target:IsDescendantOf(W) do
                    -- Điểm đích hạ thấp ngang mặt đất để chân chạm trực tiếp quả táo
                    local targetPos = target.Position
                    hum:MoveTo(targetPos)
                    touchAll(target, P.Character)

                    local dist = (Vector2.new(hrp.Position.X, hrp.Position.Z) - Vector2.new(targetPos.X, targetPos.Z)).Magnitude
                    if dist <= 1.5 or (tick() - startTime > 3.5) then
                        touchAll(target, P.Character)
                        break
                    end
                    task.wait(0.05)
                end
            end
        end
        task.wait(0.2)
    end
end)

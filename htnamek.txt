local function getClosestApple(hrp)
    local closestPart = nil
    local shortestDist = math.huge
    
    local function checkContainer(parentObj)
        for _, obj in ipairs(parentObj:GetDescendants()) do
            if obj:IsA("Model") and string.sub(obj.Name, 1, 6) == "Apple_" then
                local body = obj:FindFirstChild("Body")
                if body and body:IsA("BasePart") then
                    local dist = (hrp.Position - body.Position).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        closestPart = body
                    end
                end
            end
        end
    end
    
    checkContainer(W)
    if W.CurrentCamera then checkContainer(W.CurrentCamera) end
    
    return closestPart
end

local function touchTarget(part, char)
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    
    if firetouchinterest then
        firetouchinterest(root, part, 0)
        task.wait()
        firetouchinterest(root, part, 1)
    end
end

task.spawn(function()
    while act do
        if run and P.Character and P.Character:FindFirstChild("Humanoid") and P.Character:FindFirstChild("HumanoidRootPart") then
            local hum = P.Character.Humanoid
            local hrp = P.Character.HumanoidRootPart
            local target = getClosestApple(hrp)

            if target and target.Parent then
                local startTime = tick()
                while run and act and target and target.Parent and target:IsDescendantOf(game) do
                    local tPos = target.Position
                    hum:MoveTo(tPos)
                    touchTarget(target, P.Character)

                    -- Khi khoảng cách mặt phẳng X-Z dưới 2 stud, ép bước xuyên qua
                    local dist2D = (Vector2.new(hrp.Position.X, hrp.Position.Z) - Vector2.new(tPos.X, tPos.Z)).Magnitude
                    if dist2D <= 1.5 or (tick() - startTime > 4) then
                        touchTarget(target, P.Character)
                        break
                    end
                    task.wait(0.05)
                end
            else
                task.wait(0.3)
            end
        else
            task.wait(0.3)
        end
    end
end)

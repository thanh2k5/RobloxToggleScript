local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local LP = Players.LocalPlayer
local run, act = false, true
local MAX_DISTANCE = 300

local collectRemote = ReplicatedStorage:WaitForChild("Network", 5)
    and ReplicatedStorage.Network:WaitForChild("AppleService", 5)
    and ReplicatedStorage.Network.AppleService:WaitForChild("RF", 5)
    and ReplicatedStorage.Network.AppleService.RF:WaitForChild("Collect", 5)

local gui = Instance.new("ScreenGui")
gui.Name = "AppleCollectorGui"
gui.ResetOnSpawn = false
gui.DisplayOrder = 999999
pcall(function() gui.Parent = game:GetService("CoreGui") end)
if not gui.Parent then gui.Parent = LP:WaitForChild("PlayerGui") end

local f = Instance.new("Frame", gui)
f.Size, f.Position, f.Active, f.Draggable = UDim2.new(0, 130, 0, 40), UDim2.new(0.05, 0, 0.4, 0), true, true
f.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
Instance.new("UICorner", f).CornerRadius = UDim.new(0, 8)

local b = Instance.new("TextButton", f)
b.Size, b.Text, b.BackgroundColor3 = UDim2.new(0.72, 0, 1, 0), "OFF", Color3.fromRGB(200, 50, 50)
b.TextColor3, b.Font, b.TextSize = Color3.fromRGB(255, 255, 255), Enum.Font.GothamBold, 13
Instance.new("UICorner", b).CornerRadius = UDim.new(0, 8)

b.MouseButton1Click:Connect(function()
    run = not run
    b.Text = run and "ON" or "OFF"
    b.BackgroundColor3 = run and Color3.fromRGB(50, 180, 80) or Color3.fromRGB(200, 50, 50)
end)

local x = Instance.new("TextButton", f)
x.Size, x.Position, x.Text = UDim2.new(0.28, 0, 1, 0), UDim2.new(0.72, 0, 0, 0), "Xz"
x.BackgroundColor3, x.TextColor3 = Color3.fromRGB(45, 45, 52), Color3.fromRGB(220, 220, 220)
x.Font, x.TextSize = Enum.Font.GothamBold, 13
Instance.new("UICorner", x).CornerRadius = UDim.new(0, 8)

x.MouseButton1Click:Connect(function()
    run, act = false, false
    gui:Destroy()
end)

local function getClosestApple(hrp)
    local targetModel, targetBody = nil, nil
    local minDistance = MAX_DISTANCE
    
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Model") and string.sub(obj.Name, 1, 6) == "Apple_" then
            local body = obj:FindFirstChild("Body") or obj:FindFirstChildWhichIsA("BasePart")
            if body then
                local dist = (hrp.Position - body.Position).Magnitude
                if dist <= minDistance then
                    minDistance = dist
                    targetModel = obj
                    targetBody = body
                end
            end
        end
    end
    return targetModel, targetBody
end

local isCalling = false
local function collectApple(model)
    if not collectRemote or not model or isCalling then return end
    isCalling = true
    
    task.spawn(function()
        pcall(function()
            collectRemote:InvokeServer({model})
        end)
        task.wait(0.3)
        isCalling = false
    end)
end

task.spawn(function()
    while act do
        if run and LP.Character and LP.Character:FindFirstChild("Humanoid") and LP.Character:FindFirstChild("HumanoidRootPart") then
            local hum = LP.Character.Humanoid
            local hrp = LP.Character.HumanoidRootPart
            local model, body = getClosestApple(hrp)

            if model and body and body.Parent then
                local startTime = tick()
                while run and act and model and model.Parent and body and body.Parent do
                    hum:MoveTo(body.Position)
                    
                    local dist = (hrp.Position - body.Position).Magnitude
                    if dist <= 3.5 or (tick() - startTime > 4) then
                        collectApple(model)
                        task.wait(0.3)
                        break
                    end
                    task.wait(0.1)
                end
            else
                task.wait(0.5)
            end
        else
            task.wait(0.5)
        end
    end
end)

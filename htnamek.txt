local P, W = game:GetService("Players").LocalPlayer, game:GetService("Workspace")
local run, act = false, true

local gui = Instance.new("ScreenGui", game:GetService("CoreGui"))
local f = Instance.new("Frame", gui)
f.Size, f.Position, f.Active, f.Draggable = UDim2.new(0, 130, 0, 40), UDim2.new(0.05, 0, 0.4, 0), true, true

local b = Instance.new("TextButton", f)
b.Size, b.Text, b.BackgroundColor3 = UDim2.new(0.75, 0, 1, 0), "OFF", Color3.fromRGB(200, 50, 50)
b.MouseButton1Click:Connect(function()
    run = not run
    b.Text = run and "ON" or "OFF"
    b.BackgroundColor3 = run and Color3.fromRGB(50, 180, 80) or Color3.fromRGB(200, 50, 50)
end)

local x = Instance.new("TextButton", f)
x.Size, x.Position, x.Text = UDim2.new(0.25, 0, 1, 0), UDim2.new(0.75, 0, 0, 0), "X"
x.MouseButton1Click:Connect(function() run, act = false, false gui:Destroy() end)

local function isAppleItem(v)
    if not (v:IsA("BasePart") and v.Parent) then return false end
    if v:IsDescendantOf(P.Character or W) and v:IsDescendantOf(P.Character) then return false end
    
    -- Loại trừ mặt sàn hoặc tường trang trí quá to
    if v.Size.X > 8 or v.Size.Y > 8 or v.Size.Z > 8 then return false end
    
    local name = v.Name:lower()
    local parentName = v.Parent.Name:lower()
    
    if name:find("apple") or name:find("fruit") or name:find("táo") or name:find("drop")
       or parentName:find("apple") or parentName:find("fruit") or parentName:find("drop") then
        return true
    end
    return false
end

local function triggerTouch(part, char)
    local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Head")
    if not root then return end
    
    if firetouchinterest then
        firetouchinterest(root, part, 0)
        task.wait()
        firetouchinterest(root, part, 1)
    end
    
    local pr = part:FindFirstChildWhichIsA("ProximityPrompt", true) or (part.Parent and part.Parent:FindFirstChildWhichIsA("ProximityPrompt", true))
    if pr and fireproximityprompt then
        fireproximityprompt(pr)
    end
end

task.spawn(function()
    while act do
        if run and P.Character and P.Character:FindFirstChild("Humanoid") and P.Character:FindFirstChild("HumanoidRootPart") then
            local hum = P.Character.Humanoid
            local hrp = P.Character.HumanoidRootPart

            for _, v in ipairs(W:GetDescendants()) do
                if not run or not act then break end
                
                if isAppleItem(v) then
                    local startTime = tick()
                    while run and act and v and v.Parent and isAppleItem(v) do
                        hum:MoveTo(v.Position)
                        triggerTouch(v, P.Character)
                        
                        -- Chạy vào sát tâm vật phẩm dưới 2 stud là đã nhặt thành công
                        if (hrp.Position - v.Position).Magnitude <= 2 or (tick() - startTime > 4) then
                            triggerTouch(v, P.Character)
                            break
                        end
                        task.wait(0.05)
                    end
                end
            end
        end
        task.wait(0.3)
    end
end)

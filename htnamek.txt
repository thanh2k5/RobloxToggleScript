-- 1. TẠO UI DEBUG PHÙ HỢP CHO DELTA MOBILE
local PlayerGui = LP:WaitForChild("PlayerGui", 10)
local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "AdvancedDebugGui"
ScreenGui.ResetOnSpawn = false
ScreenGui.DisplayOrder = 999999
ScreenGui.Parent = PlayerGui

local Frame = Instance.new("Frame", ScreenGui)
Frame.Size = UDim2.new(0, 300, 0, 220)
Frame.Position = UDim2.new(0.5, -150, 0.15, 0)
Frame.BackgroundColor3 = Color3.fromRGB(18, 18, 22)
Frame.BorderSizePixel = 0
Frame.Active = true

Instance.new("UICorner", Frame).CornerRadius = UDim.new(0, 8)

-- Kéo thả cảm ứng trên điện thoại
local dragging, dragInput, dragStart, startPos
Frame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = Frame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
Frame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        Frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- 2. THEO DÕI VẬT LÝ VÀ CẤU TRÚC MODEL TÁO
local tracked = {}

task.spawn(function()
    while isAlive do
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA("Model") and string.sub(obj.Name, 1, 6) == "Apple_" and not tracked[obj] then
                    local body = obj:FindFirstChild("Body")
                    if body and body:IsA("BasePart") then
                        local dist = (hrp.Position - body.Position).Magnitude
                        if dist <= 12 then
                            tracked[obj] = true
                            
                            -- In chi tiết các thành phần trong Model táo
                            log("[PHAT HIEN TAO GAN BAN]", Color3.fromRGB(255, 200, 80))
                            log("• Model: " .. obj.Name)
                            log("• FullPath: " .. obj:GetFullName())
                            
                            -- Kiểm tra các con bên trong Model
                            local childrenNames = {}
                            for _, c in ipairs(obj:GetChildren()) do
                                table.insert(childrenNames, c.Name .. " (" .. c.ClassName .. ")")
                            end
                            log("• Con ben trong: " .. table.concat(childrenNames, ", "))
                            
                            -- Kiểm tra Attributes
                            local attrs = obj:GetAttributes()
                            local attrCount = 0
                            for k, v in pairs(attrs) do
                                attrCount = attrCount + 1
                                log("• Attr: " .. tostring(k) .. " = " .. tostring(v))
                            end
                            
                            -- Lắng nghe xem Model hoặc Body biến mất thế nào
                            obj.AncestryChanged:Connect(function(_, parent)
                                if not parent and isAlive then
                                    log("[MODEL TAO DA BI NHAT/XOA]", Color3.fromRGB(80, 255, 120))
                                    log("• Model da bi Destroy(): " .. obj.Name)
                                end
                            end)
                        end
                    end
                end
            end
        end
        task.wait(0.2)
    end
end)

-- 3. HOOK MỌI LỆNH GỬI LÊN SERVER (FIRESERVER / INVOKESERVER)
local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    local args = {...}
    
    if isAlive and (method == "FireServer" or method == "InvokeServer") then
        local rName = self.Name
        -- Lọc bỏ các remote hệ thống mặc định của Roblox
        if rName ~= "UpdatePlayerBlockList" and rName ~= "CharacterLoaded" and not rName:find("Chat") then
            log("[REMOTE GOI]: " .. rName, Color3.fromRGB(255, 130, 220))
            log("• Method: " .. method .. " | Path: " .. self:GetFullName())
            for i, arg in ipairs(args) do
                local valStr = tostring(arg)
                if typeof(arg) == "Instance" then
                    valStr = arg:GetFullName()
                elseif typeof(arg) == "table" then
                    valStr = "{table: " .. #arg .. " items}"
                end
                log("  arg[" .. i .. "] (" .. typeof(arg) .. "): " .. valStr)
            end
        end
    end
    
    return oldNamecall(self, ...)
end)

local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local LP = Players.LocalPlayer

print("--- [DEBUG AUTO COLLECT STARTED] ---")
print(">> Vui lòng đi lại gần và NHẶT 1 quả táo bằng tay...")

-- 1. Quét các Part quanh nhân vật trong bán kính 10 studs để phát hiện Part bị xoá khi nhặt
local trackedParts = {}

task.spawn(function()
    while true do
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            for _, v in ipairs(Workspace:GetDescendants()) do
                if v:IsA("BasePart") and not v:IsDescendantOf(char) then
                    local dist = (hrp.Position - v.Position).Magnitude
                    if dist <= 12 and not trackedParts[v] then
                        trackedParts[v] = true
                        
                        -- Lắng nghe xem part này có bị nhặt (Destroyed / Removed) không
                        v.AncestryChanged:Connect(function(_, parent)
                            if not parent then
                                warn("=================================")
                                print("[PHÁT HIỆN VẬT PHẨM VỪA ĐƯỢC NHẶT!]")
                                print("• Tên Part:", v.Name)
                                print("• Class:", v.ClassName)
                                print("• Model cha (Parent):", v.Parent and v.Parent.Name or "None")
                                print("• Anchored:", v.Anchored)
                                print("• CanCollide:", v.CanCollide)
                                print("• Size:", tostring(v.Size))
                                
                                local p = v.Parent
                                local path = v.Name
                                while p and p ~= Workspace do
                                    path = p.Name .. " -> " .. path
                                    p = p.Parent
                                end
                                print("• Đường dẫn (Path):", "Workspace -> " .. path)
                                warn("=================================")
                            end
                        end)
                    end
                end
            end
        end
        task.wait(0.2)
    end
end)

-- 2. Hook lắng nghe RemoteEvent / RemoteFunction nếu game dùng cơ chế gửi lệnh nhặt
local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    local args = {...}
    
    if (method == "FireServer" or method == "InvokeServer") then
        local remoteName = self.Name:lower()
        if remoteName:find("pickup") or remoteName:find("collect") or remoteName:find("item") or remoteName:find("drop") or remoteName:find("apple") or remoteName:find("fruit") then
            warn(">>> PHÁT HIỆN REMOTE NHẶT:", self:GetFullName())
            print("Arguments gửi lên:")
            for i, arg in ipairs(args) do
                print(string.format("  [%d]: %s (%s)", i, tostring(arg), typeof(arg)))
            end
        end
    end
    
    return oldNamecall(self, ...)
end)
